cmake_minimum_required(VERSION 3.16)
project(s5_audio_offline LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# A Windows GNU host may not ship libssp; these only affect this offline build.
if(WIN32)
  set(OPUS_STACK_PROTECTOR OFF CACHE BOOL "" FORCE)
  set(OPUS_FORTIFY_SOURCE OFF CACHE BOOL "" FORCE)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../audio/S5Audio.cmake")
s5_add_audio_core()
find_package(Threads REQUIRED)
add_executable(s5_audio_offline_test audio_offline_test.cpp)
target_link_libraries(s5_audio_offline_test PRIVATE s5_audio_core Threads::Threads)
enable_testing()
add_test(NAME s5_audio_offline COMMAND s5_audio_offline_test)
