cmake_minimum_required(VERSION 3.5.0)
project(uus5 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(OHOS_ARCH STREQUAL "arm64-v8a" OR CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  set(S5_RUST_TARGET aarch64-unknown-linux-ohos)
else()
  set(S5_RUST_TARGET x86_64-unknown-linux-ohos)
endif()
if(NOT DEFINED S3B_RUST)
  set(S3B_RUST "${CMAKE_CURRENT_SOURCE_DIR}/../../../../rust/target/${S5_RUST_TARGET}/release/libuus5.a")
endif()
if(NOT EXISTS "${S3B_RUST}")
  message(FATAL_ERROR "Run node scripts/build.cjs native first")
endif()
add_library(s3ccore STATIC IMPORTED)
set_target_properties(s3ccore PROPERTIES IMPORTED_LOCATION "${S3B_RUST}")
add_library(s3c SHARED napi_init.cpp video.cpp)
target_compile_options(s3c PRIVATE -Wall -Wextra)
target_link_options(s3c PRIVATE -Wl,--no-undefined)
target_link_libraries(s3c PUBLIC s3ccore libace_napi.z.so libasset_ndk.z.so libhilog_ndk.z.so libnative_media_codecbase.so libnative_media_core.so libnative_media_vdec.so libnative_window.so dl m)

include("${CMAKE_CURRENT_SOURCE_DIR}/audio/S5Audio.cmake")
s5_add_advanced_audio(s3c)

include("${CMAKE_CURRENT_SOURCE_DIR}/pc_host/UnifiedHost.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/files/Files.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/microphone/Microphone.cmake")
